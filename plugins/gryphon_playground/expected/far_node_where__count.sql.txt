SELECT "tap_edge"."from_entity_id" AS "_g_col_0", COUNT(T5."to_entity_id") AS "_g_agg_0"
FROM "tap_edge"
INNER JOIN "tap_entity" ON ("tap_edge"."entity_id" = "tap_entity"."id")
INNER JOIN "tap_entity" T3 ON ("tap_edge"."from_entity_id" = T3."id")
INNER JOIN "tap_entity" T4 ON ("tap_edge"."to_entity_id" = T4."id")
INNER JOIN "tap_edge" T5 ON (T4."id" = T5."from_entity_id")
INNER JOIN "tap_entity" T6 ON (T5."to_entity_id" = T6."id")
INNER JOIN "gryphon_playground__pg_node" ON (T6."id" = "gryphon_playground__pg_node"."entity_id")
WHERE ("tap_entity"."deleted_at" IS NULL AND "tap_edge"."edge_type" = %s AND T3."entity_type" = %s AND T5."edge_type" = %s AND T6."entity_type" = %s AND T4."entity_type" = %s AND "gryphon_playground__pg_node"."kind" = %s)
GROUP BY 1
ORDER BY 1 ASC
