-- statement 1 · stage: single-hop
SELECT "tap_edge"."from_entity_id" AS "from_entity_id", "tap_edge"."to_entity_id" AS "to_entity_id", "tap_edge"."entity_id" AS "entity_id"
FROM "tap_edge"
INNER JOIN "tap_entity" ON ("tap_edge"."entity_id" = "tap_entity"."id")
INNER JOIN "tap_entity" T3 ON ("tap_edge"."from_entity_id" = T3."id")
INNER JOIN "tap_entity" T4 ON ("tap_edge"."to_entity_id" = T4."id")
WHERE ("tap_entity"."deleted_at" IS NULL AND "tap_edge"."edge_type" = %s AND T3."entity_type" = %s AND T4."entity_type" = %s)
ORDER BY "tap_entity"."created_at" DESC
