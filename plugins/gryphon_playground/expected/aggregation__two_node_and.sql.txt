SELECT "tap_edge"."from_entity_id" AS "_g_col_0", COUNT("tap_edge"."to_entity_id") AS "_g_agg_0"
FROM "tap_edge"
INNER JOIN "tap_entity" ON ("tap_edge"."entity_id" = "tap_entity"."id")
INNER JOIN "tap_entity" T3 ON ("tap_edge"."from_entity_id" = T3."id")
INNER JOIN "tap_entity" T4 ON ("tap_edge"."to_entity_id" = T4."id")
INNER JOIN "gryphon_playground__pg_hub" ON (T3."id" = "gryphon_playground__pg_hub"."entity_id")
INNER JOIN "gryphon_playground__pg_node" ON (T4."id" = "gryphon_playground__pg_node"."entity_id")
WHERE ("tap_entity"."deleted_at" IS NULL AND "tap_edge"."edge_type" = %s AND T3."entity_type" = %s AND T4."entity_type" = %s AND "gryphon_playground__pg_hub"."severity_score" = %s AND "gryphon_playground__pg_node"."severity_score" > %s)
GROUP BY 1
ORDER BY 1 ASC
